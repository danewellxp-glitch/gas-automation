Axios Instance (driverApi.js):
javascriptimport axios from 'axios';

const API_BASE = 'http://192.168.10.156:8000/api';

const api = axios.create({
  baseURL: API_BASE,
  headers: {
    'Content-Type': 'application/json'
  }
});

// Interceptor para JWT
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Interceptor para erros
api.interceptors.response.use(
  response => response,
  error => {
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/driver/login';
    }
    return Promise.reject(error);
  }
);

export const driverApi = {
  login: (username, email, password) => 
    api.post('/auth/login', { username, email, password }),
  
  getMe: () => api.get('/drivers/me'),
  getMyStats: () => api.get('/drivers/me/stats'),
  updateStatus: (status) => 
    api.put('/drivers/me/status', { status }),
  updateLocation: (lat, lng) => 
    api.put('/drivers/me/location', { latitude: lat, longitude: lng }),
  
  getMyDeliveries: (status) => {
    const params = status ? { status } : {};
    return api.get('/drivers/me/deliveries', { params });
  },
  
  updateDeliveryStatus: (id, status, notes = '') => 
    api.put(`/drivers/deliveries/${id}/status`, { status, notes }),
  
  reportProblem: (id, problem_type, description) => 
    api.post(`/drivers/deliveries/${id}/problem`, { 
      problem_type, 
      description 
    })
};

export default api;
WebSocket Hook (useWebSocketDriver.js):
javascriptimport { useEffect, useState } from 'react';

export function useWebSocketDriver() {
  const [connected, setConnected] = useState(false);
  const [lastMessage, setLastMessage] = useState(null);
  
  useEffect(() => {
    const token = localStorage.getItem('token');
    if (!token) return;
    
    const ws = new WebSocket(
      `ws://192.168.10.156:8000/ws/dashboard?token=${token}`
    );
    
    ws.onopen = () => {
      console.log('WebSocket conectado');
      setConnected(true);
    };
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      console.log('WebSocket message:', data);
      setLastMessage(data);
      
      // Notificações
      if (data.type === 'delivery_assigned') {
        showNotification('Nova entrega disponível!');
        playSound();
      }
    };
    
    ws.onerror = (error) => {
      console.error('WebSocket error:', error);
    };
    
    ws.onclose = () => {
      console.log('WebSocket desconectado');
      setConnected(false);
    };
    
    return () => ws.close();
  }, []);
  
  return { connected, lastMessage };
}

function showNotification(message) {
  if ('Notification' in window && Notification.permission === 'granted') {
    new Notification('Gas Automation', { body: message });
  }
}

function playSound() {
  const audio = new Audio('/notification.mp3');
  audio.play().catch(e => console.log('Audio play failed:', e));
}
