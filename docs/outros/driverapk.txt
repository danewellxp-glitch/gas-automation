Transformar o dashboard do motorista em um app Android (APK) usando Expo - a forma mais fácil!

### 📄 src/services/api.js

```javascript
import AsyncStorage from '@react-native-async-storage/async-storage';
import axios from 'axios';

// ⚠️ ALTERE AQUI PARA SEU IP/SERVIDOR
const API_URL = 'http://192.168.10.156:8000';

const api = axios.create({
  baseURL: API_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Adicionar token automaticamente
api.interceptors.request.use(async (config) => {
  const token = await AsyncStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Tratar erros de autenticação
api.interceptors.response.use(
  response => response,
  async error => {
    if (error.response?.status === 401) {
      await AsyncStorage.removeItem('token');
      await AsyncStorage.removeItem('user');
    }
    return Promise.reject(error);
  }
);

// Autenticação
export const authAPI = {
  login: async (phone, password) => {
    const formData = new URLSearchParams();
    formData.append('username', phone);
    formData.append('password', password);
    
    const response = await api.post('/api/auth/login', formData, {
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' }
    });
    
    return response.data;
  }
};

// APIs do Motorista
export const driverAPI = {
  getMe: () => api.get('/api/drivers/me'),
  
  getActiveDeliveries: () => 
    api.get('/api/drivers/me/deliveries/active'),
  
  getAvailableDeliveries: () => 
    api.get('/api/drivers/deliveries/available'),
  
  acceptDelivery: (id) => 
    api.post(`/api/drivers/deliveries/${id}/accept`),
  
  updateDeliveryStatus: (id, status) => 
    api.put(`/api/drivers/deliveries/${id}/status`, { status }),
  
  updateStatus: (status) => 
    api.put('/api/drivers/me/status', { status }),
  
  getStats: () => 
    api.get('/api/drivers/me/stats/today'),
  
  getTimeWorked: () => 
    api.get('/api/drivers/me/time-worked'),
};

export default api;
```

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  ActivityIndicator,
  Alert,
  KeyboardAvoidingView,
  Platform,
  StatusBar,
} from 'react-native';
import { Phone, Lock } from 'lucide-react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { authAPI } from '../services/api';
import { COLORS } from '../utils/constants';

export default function LoginScreen({ navigation }) {
  const [phone, setPhone] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);

  const formatPhone = (value) => {
    const numbers = value.replace(/\D/g, '');
    if (numbers.length <= 2) return numbers;
    if (numbers.length <= 7) return `(${numbers.slice(0, 2)}) ${numbers.slice(2)}`;
    return `(${numbers.slice(0, 2)}) ${numbers.slice(2, 7)}-${numbers.slice(7, 11)}`;
  };

  const handleLogin = async () => {
    if (!phone || !password) {
      Alert.alert('Erro', 'Preencha todos os campos');
      return;
    }

    setLoading(true);
    try {
      const data = await authAPI.login(phone, password);
      
      if (data.role !== 'driver') {
        Alert.alert('Erro', 'Acesso negado: você não é um motorista');
        setLoading(false);
        return;
      }

      await AsyncStorage.setItem('token', data.access_token);
      await AsyncStorage.setItem('user', JSON.stringify(data));
      
      navigation.replace('Dashboard');
    } catch (error) {
      Alert.alert('Erro', error.response?.data?.detail || 'Erro ao fazer login');
    } finally {
      setLoading(false);
    }
  };

  return (
    <KeyboardAvoidingView 
      style={styles.container} 
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
    >
      <StatusBar barStyle="light-content" backgroundColor={COLORS.background} />
      
      <View style={styles.content}>
        {/* Logo */}
        <View style={styles.logoContainer}>
          <View style={styles.logo}>
            <Text style={styles.logoText}>⚡</Text>
          </View>
          <Text style={styles.title}>Gas Automation</Text>
          <Text style={styles.subtitle}>Portal do Motorista</Text>
        </View>

        {/* Formulário */}
        <View style={styles.form}>
          {/* Telefone */}
          <View style={styles.inputContainer}>
            <Phone color={COLORS.textSecondary} size={20} style={styles.inputIcon} />
            <TextInput
              style={styles.input}
              placeholder="(00) 00000-0000"
              placeholderTextColor={COLORS.textSecondary}
              value={phone}
              onChangeText={(text) => setPhone(formatPhone(text))}
              keyboardType="phone-pad"
              maxLength={15}
              editable={!loading}
            />
          </View>

          {/* Senha */}
          <View style={styles.inputContainer}>
            <Lock color={COLORS.textSecondary} size={20} style={styles.inputIcon} />
            <TextInput
              style={styles.input}
              placeholder="Senha"
              placeholderTextColor={COLORS.textSecondary}
              value={password}
              onChangeText={setPassword}
              secureTextEntry
              editable={!loading}
            />
          </View>

          {/* Botão Login */}
          <TouchableOpacity
            style={[styles.button, loading && styles.buttonDisabled]}
            onPress={handleLogin}
            disabled={loading}
          >
            {loading ? (
              <ActivityIndicator color="#FFF" />
            ) : (
              <Text style={styles.buttonText}>Entrar</Text>
            )}
          </TouchableOpacity>
        </View>
      </View>
    </KeyboardAvoidingView>
  );
}

import React, { useState, useEffect, useCallback } from 'react';
import {
  View,
  Text,
  ScrollView,
  TouchableOpacity,
  StyleSheet,
  RefreshControl,
  StatusBar,
  Alert,
} from 'react-native';
import { 
  Package, 
  Star, 
  Clock, 
  DollarSign,
  Menu,
  LogOut,
} from 'lucide-react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { driverAPI } from '../services/api';
import { COLORS, DRIVER_STATUS } from '../utils/constants';
import DeliveryCard from '../components/DeliveryCard';
import StatsCard from '../components/StatsCard';
import StatusToggle from '../components/StatusToggle';

export default function DashboardScreen({ navigation }) {
  const [driver, setDriver] = useState(null);
  const [activeDeliveries, setActiveDeliveries] = useState([]);
  const [availableDeliveries, setAvailableDeliveries] = useState([]);
  const [stats, setStats] = useState({ today: 0, rating: 5.0, avgTime: 0, earnings: 0 });
  const [timeWorked, setTimeWorked] = useState({ today: '0h 0m', week: '0h 0m' });
  const [refreshing, setRefreshing] = useState(false);

  const loadData = useCallback(async () => {
    try {
      const [
        driverRes,
        activeRes,
        availableRes,
        statsRes,
        timeRes
      ] = await Promise.all([
        driverAPI.getMe(),
        driverAPI.getActiveDeliveries(),
        driverAPI.getAvailableDeliveries(),
        driverAPI.getStats(),
        driverAPI.getTimeWorked(),
      ]);

      setDriver(driverRes.data);
      setActiveDeliveries(activeRes.data);
      setAvailableDeliveries(availableRes.data);
      setStats(statsRes.data);
      setTimeWorked(timeRes.data);
    } catch (error) {
      console.error('Erro ao carregar dados:', error);
      if (error.response?.status === 401) {
        handleLogout();
      }
    }
  }, []);

  useEffect(() => {
    loadData();
    const interval = setInterval(loadData, 30000);
    return () => clearInterval(interval);
  }, [loadData]);

  const onRefresh = async () => {
    setRefreshing(true);
    await loadData();
    setRefreshing(false);
  };

  const handleStatusChange = async (newStatus) => {
    try {
      await driverAPI.updateStatus(newStatus);
      setDriver({ ...driver, status: newStatus });
    } catch (error) {
      Alert.alert('Erro', 'Não foi possível atualizar o status');
    }
  };

  const handleAcceptDelivery = async (deliveryId) => {
    try {
      await driverAPI.acceptDelivery(deliveryId);
      await loadData();
      Alert.alert('Sucesso', 'Entrega aceita!');
    } catch (error) {
      Alert.alert('Erro', 'Não foi possível aceitar a entrega');
    }
  };

  const handleLogout = async () => {
    await AsyncStorage.clear();
    navigation.replace('Login');
  };

  return (
    <View style={styles.container}>
      <StatusBar barStyle="light-content" backgroundColor={COLORS.background} />
      
      {/* Header */}
      <View style={styles.header}>
        <View style={styles.headerLeft}>
          <View style={styles.avatar}>
            <Text style={styles.avatarText}>
              {driver?.name?.charAt(0) || 'D'}
            </Text>
          </View>
          <View>
            <Text style={styles.driverName}>{driver?.name || 'Motorista'}</Text>
            <View style={styles.statusIndicator}>
              <View style={[styles.statusDot, { 
                backgroundColor: DRIVER_STATUS[driver?.status]?.color 
              }]} />
              <Text style={styles.statusText}>
                {DRIVER_STATUS[driver?.status]?.label || 'Offline'}
              </Text>
            </View>
          </View>
        </View>
        <TouchableOpacity onPress={handleLogout} style={styles.logoutButton}>
          <LogOut color={COLORS.textSecondary} size={24} />
        </TouchableOpacity>
      </View>

      <ScrollView
        style={styles.scrollView}
        refreshControl={
          <RefreshControl refreshing={refreshing} onRefresh={onRefresh} />
        }
      >
        {/* Status Toggle */}
        <StatusToggle 
          currentStatus={driver?.status} 
          onStatusChange={handleStatusChange}
        />

        {/* Stats Cards */}
        <View style={styles.statsGrid}>
          <StatsCard
            icon={Package}
            value={stats.today}
            label="Entregas Hoje"
            color="#10B981"
          />
          <StatsCard
            icon={Star}
            value={stats.rating.toFixed(1)}
            label="Avaliação"
            color="#F59E0B"
          />
          <StatsCard
            icon={Clock}
            value={`${stats.avgTime}m`}
            label="Tempo Médio"
            color="#3B82F6"
          />
          <StatsCard
            icon={DollarSign}
            value={`R$ ${stats.earnings}`}
            label="Ganhos"
            color="#A855F7"
          />
        </View>

        {/* Tempo Trabalhado */}
        <View style={styles.timeCard}>
          <Text style={styles.sectionTitle}>⏱️ Tempo Trabalhado</Text>
          <View style={styles.timeGrid}>
            <View>
              <Text style={styles.timeLabel}>Hoje</Text>
              <Text style={styles.timeValue}>{timeWorked.today}</Text>
            </View>
            <View>
              <Text style={styles.timeLabel}>Esta Semana</Text>
              <Text style={styles.timeValue}>{timeWorked.week}</Text>
            </View>
          </View>
        </View>

        {/* Entregas Ativas */}
        {activeDeliveries.length > 0 && (
          <View style={styles.section}>
            <Text style={styles.sectionTitle}>
              🚚 Entregas Ativas ({activeDeliveries.length})
            </Text>
            {activeDeliveries.map((delivery) => (
              <DeliveryCard
                key={delivery.id}
                delivery={delivery}
                onPress={() => navigation.navigate('DeliveryDetail', { delivery })}
                active
              />
            ))}
          </View>
        )}

        {/* Entregas Disponíveis */}
        {availableDeliveries.length > 0 && (
          <View style={styles.section}>
            <Text style={styles.sectionTitle}>
              📦 Entregas Disponíveis ({availableDeliveries.length})
            </Text>
            {availableDeliveries.map((delivery) => (
              <DeliveryCard
                key={delivery.id}
                delivery={delivery}
                onAccept={() => handleAcceptDelivery(delivery.id)}
                available
              />
            ))}
          </View>
        )}

        {/* Mensagem quando vazio */}
        {activeDeliveries.length === 0 && availableDeliveries.length === 0 && (
          <View style={styles.emptyState}>
            <Package color={COLORS.textSecondary} size={64} />
            <Text style={styles.emptyTitle}>Nenhuma entrega no momento</Text>
            <Text style={styles.emptyText}>
              Fique conectado para receber novas entregas
            </Text>
          </View>
        )}
      </ScrollView>
    </View>
  );
}

import React from 'react';
import { View, Text, TouchableOpacity, StyleSheet } from 'react-native';
import { COLORS, DRIVER_STATUS } from '../utils/constants';

export default function StatusToggle({ currentStatus, onStatusChange }) {

    <View style={styles.container}>

  );
}

### 📄 App.js (Navegação Principal)

```javascript
import React, { useEffect, useState } from 'react';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { ActivityIndicator, View } from 'react-native';

import LoginScreen from './src/screens/LoginScreen';
import DashboardScreen from './src/screens/DashboardScreen';
import DeliveryDetailScreen from './src/screens/DeliveryDetailScreen';
import { COLORS } from './src/utils/constants';

const Stack = createNativeStackNavigator();

export default function App() {
  const [isLoading, setIsLoading] = useState(true);
  const [isLoggedIn, setIsLoggedIn] = useState(false);

  useEffect(() => {
    checkAuth();
  }, []);

  const checkAuth = async () => {
    try {
      const token = await AsyncStorage.getItem('token');
      setIsLoggedIn(!!token);
    } catch (error) {
      console.error('Erro ao verificar autenticação:', error);
    } finally {
      setIsLoading(false);
    }
  };

  if (isLoading) {
    return (
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: COLORS.background }}>
        <ActivityIndicator size="large" color={COLORS.primary} />
      </View>
    );
  }

  return (
    <NavigationContainer>
      <Stack.Navigator
        initialRouteName={isLoggedIn ? 'Dashboard' : 'Login'}
        screenOptions={{
          headerShown: false,
          contentStyle: { backgroundColor: COLORS.background },
        }}
      >
        <Stack.Screen name="Login" component={LoginScreen} />
        <Stack.Screen name="Dashboard" component={DashboardScreen} />
        <Stack.Screen 
          name="DeliveryDetail" 
          component={DeliveryDetailScreen}
          options={{
            headerShown: true,
            title: 'Detalhes da Entrega',
            headerStyle: { backgroundColor: COLORS.card },
            headerTintColor: COLORS.text,
          }}
        />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
```

## 🐛 TROUBLESHOOTING

### Erro de rede / CORS

```javascript
// No seu backend FastAPI, adicione:
from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
```

### APK não conecta ao servidor

1. Certifique-se de que o celular está na mesma rede WiFi
2. Use o IP local (192.168.x.x), não localhost
3. Teste a URL no navegador do celular primeiro
